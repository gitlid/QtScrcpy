find_package(Qt${QT_DESIRED_VERSION} REQUIRED COMPONENTS WebEngineWidgets WebChannel)
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
set(QSC_KEYMAPPER_NPM_CACHE "${CMAKE_BINARY_DIR}/keymapper-npm" CACHE PATH "Pinned npm archive cache")
set(ASSETS "${CMAKE_CURRENT_BINARY_DIR}/assets")
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/ci/prepare_keymapper.py
    --root ${CMAKE_SOURCE_DIR} --output ${ASSETS} --cache ${QSC_KEYMAPPER_NPM_CACHE}
    RESULT_VARIABLE prepared)
if(NOT prepared EQUAL 0)
    message(FATAL_ERROR "Preparing offline ScrcpyKeyMapper assets failed")
endif()
add_library(qsc_keymapper STATIC webkeymapdialog.cpp webkeymapdialog.h "${ASSETS}/keymapper_assets.qrc")
set_target_properties(qsc_keymapper PROPERTIES AUTOMOC ON AUTORCC ON)
target_link_libraries(qsc_keymapper PUBLIC Qt${QT_DESIRED_VERSION}::WebEngineWidgets Qt${QT_DESIRED_VERSION}::WebChannel)
target_include_directories(qsc_keymapper PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(qsc_keymapper PUBLIC QSC_WITH_KEYMAPPER)
if(MSVC)
    target_compile_options(qsc_keymapper PRIVATE /utf-8)
endif()
if(BUILD_TESTING)
    add_executable(keymapper_smoke ${CMAKE_SOURCE_DIR}/tests/keymapper_smoke.cpp)
    target_link_libraries(keymapper_smoke PRIVATE qsc_keymapper)
    add_test(NAME keymapper_native_bridge COMMAND keymapper_smoke)
    set_tests_properties(keymapper_native_bridge PROPERTIES TIMEOUT 45)
endif()
