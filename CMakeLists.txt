cmake_minimum_required(VERSION 3.19 FATAL_ERROR)
project(all)
include(CTest)
add_subdirectory(QtScrcpy)
target_sources(QtScrcpy PRIVATE QtScrcpy/ui/keymapeditor.cpp QtScrcpy/ui/keymapeditor.h QtScrcpy/ui/keymapdocument.h QtScrcpy/ui/macroexecutionoptions.h)
option(QSC_WITH_KEYMAPPER "Embed the offline upstream ScrcpyKeyMapper editor" ON)
if(QSC_WITH_KEYMAPPER)
    if(NOT QT_DESIRED_VERSION)
        find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core)
        set(QT_DESIRED_VERSION ${QT_VERSION_MAJOR})
    endif()
    add_subdirectory(QtScrcpy/keymapper)
    target_link_libraries(QtScrcpy PRIVATE qsc_keymapper)
endif()
if(BUILD_TESTING AND QSC_BUILD_ACTION_MACRO_TESTS)
    add_subdirectory(tests)
    add_subdirectory(tests/v2layout)
endif()
if(TARGET actionmacro_integration AND QSC_WITH_KEYMAPPER)
    target_link_libraries(actionmacro_integration PRIVATE qsc_keymapper)
endif()
