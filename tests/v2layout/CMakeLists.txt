find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets Network)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 11)
add_executable(keymap_layout_test keymap_layout_test.cpp ../../QtScrcpy/ui/keymapeditor.cpp ../../QtScrcpy/ui/keymapeditor.h)
target_include_directories(keymap_layout_test PRIVATE ../../QtScrcpy/ui)
target_link_libraries(keymap_layout_test PRIVATE QtScrcpyCore Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::Network)
if(MSVC)
    target_compile_options(keymap_layout_test PRIVATE /utf-8)
endif()
foreach(case_name IN ITEMS portrait landscape resized)
    add_test(NAME keymap_layout_${case_name} COMMAND keymap_layout_test ${case_name})
    set_tests_properties(keymap_layout_${case_name} PROPERTIES TIMEOUT 15)
endforeach()
